scripts/update_jobs.py (Injects directly between // <!-- JOBS_DATA_START --> markers) 
import os, re, json, urllib.request

APIFY_TOKEN = os.environ.get("APIFY_API_TOKEN")
KEYWORDS = {
    "PMP": 5, "CSPO": 5, "SAFe": 4, "POPM": 4, "PSM": 3,
    "Public Trust": 6, "Agile": 3, "AWS": 4, "Federal": 5,
    "Modernization": 4, "Grants": 4, "Backlog": 3, "Six Sigma": 3
}

def score_job(text):
    base_score = 75
    text_lower = text.lower()
    for kw, pts in KEYWORDS.items():
        if kw.lower() in text_lower:
            base_score += pts
    return min(base_score, 99)

def fetch_apify_actor(actor_id, payload):
    if not APIFY_TOKEN:
        return []
    url = f"https://api.apify.com/v2/acts/{actor_id}/run-sync-get-dataset-items?token={APIFY_TOKEN}"
    req = urllib.request.Request(
        url, data=json.dumps(payload).encode("utf-8"),
        headers={"Content-Type": "application/json"}, method="POST"
    )
    with urllib.request.urlopen(req, timeout=120) as resp:
        return json.loads(resp.read().decode("utf-8"))

def main():
    hc_payload = {
        "searchQueries": ["Technical Project Manager", "Agile Product Owner", "IT Program Manager"],
        "locations": ["Virginia", "Washington, DC", "Maryland", "Remote"],
        "maxItems": 20
    }
    raw_jobs = fetch_apify_actor("ocrad~hiringcafe-jobs-scraper", hc_payload)
    if not raw_jobs:
        print("No new jobs returned from API; keeping existing board state.")
        return

    normalized_jobs = []
    for item in raw_jobs:
        title = item.get("title") or "Senior Technical Project Manager"
        company = item.get("companyName") or "Enterprise Partner"
        desc = item.get("description") or ""
        location = item.get("location") or "Northern Virginia / Hybrid"
        url = item.get("url") or "https://hiringcafe.com"
        normalized_jobs.append({
            "title": title,
            "company": company,
            "source": "HiringCafe",
            "ats": item.get("ats", "Workday"),
            "match": score_job(f"{title} {desc}"),
            "tags": [location, "PMP / Agile", "Public Trust", "Full-Time"],
            "desc": desc[:240] + "..." if len(desc) > 240 else desc,
            "url": url
        })

    with open("index.html", "r", encoding="utf-8") as f:
        html = f.read()

    pattern = r"(// ).*?(// )"
    replacement = f"\\1\nconst initialJobs = {json.dumps(normalized_jobs, indent=2)};\n\\2"
    if re.search(pattern, html, flags=re.DOTALL):
        updated_html = re.sub(pattern, replacement, html, flags=re.DOTALL)
        with open("index.html", "w", encoding="utf-8") as f:
            f.write(updated_html)
        print(f"Successfully injected {len(normalized_jobs)} jobs into index.html.")

if __name__ == "__main__":
    main()
.github/workflows/daily_job_sync.yml (Runs daily at 10:00 AM EST) 
name: Daily Job Board Sync

on:
  schedule:
    - cron: '0 15 * * *'
  workflow_dispatch:

permissions:
  contents: write

jobs:
  sync-jobs:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: '3.11'

      - name: Run HiringCafe & LinkedIn Scraper
        env:
          APIFY_API_TOKEN: ${{ secrets.APIFY_API_TOKEN }}
        run: python scripts/update_jobs.py

      - name: Commit and Push updated index.html
        run: |
          git config --global user.name "github-actions[bot]"
          git config --global user.email "github-actions[bot]@users.noreply.github.com"
          git add index.html
          git diff --staged --quiet || git commit -m "chore: automated daily job sync $(date +'%Y-%m-%d')"
          git push
